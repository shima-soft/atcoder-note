# ABC461D - Count Subgrid Sum = K

def count_eq(arr, k):
  n = len(arr)
  r1 = r2 = 0
  res = 0
  for l in range(n - 1):
    r1 = max(r1, l + 1)
    r2 = max(r2, l + 1)
    while r1 < n and arr[r1] - arr[l] < k:
      r1 += 1
    while r2 < n and arr[r2] - arr[l] <= k:
      r2 += 1
    res += r2 - r1
  return res

def solve():
  H, W, K = map(int, input().split())
  S = [input().strip() for _ in range(H)]
  ans = 0
  for u in range(H):
    col_sum = [0] * W
    for d in range(u, H):
      row = S[d]
      for j in range(W):
        if row[j] == '1':
          col_sum[j] += 1
      B = [0] * (W + 1)
      for j in range(W):
        B[j + 1] = B[j] + col_sum[j]
      ans += count_eq(B, K)
  print(ans)
実装のポイント
1
col_sum は行範囲を拡張するたびにインクリメント。毎回ゼロから再計算しない。
2
各 (u,d) で B を O(W) で構築 → count_eq で O(W) → 計 O(W) の定数倍。
3
K=0 の場合も尺取り法で正しく動作（条件式は < K と ≤ K）。
