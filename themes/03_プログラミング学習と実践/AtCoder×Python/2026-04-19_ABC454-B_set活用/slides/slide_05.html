set() と len() の組み合わせで2つの条件を同時に判定する

abc454_b.py
N, M = map(int, input().split())
F = list(map(int, input().split()))

# Q1: 全員バラバラな服？
if len(set(F)) == N:
    print("Yes")
else:
    print("No")

# Q2: 全種類カバーしてる？
if len(set(F)) == M:
    print("Yes")
else:
    print("No")
例1: N=3, M=4, F=[1,2,4]
set(F)={1,2,4} → len=3
Q1: 3==3 → Yes Q2: 3==4 → No
例2: N=4, M=2, F=[1,2,1,2]
set(F)={1,2} → len=2
Q1: 2==4 → No Q2: 2==2 → Yes
例3: N=4, M=4, F=[1,3,2,1]
set(F)={1,2,3} → len=3
Q1: 3==4 → No Q2: 3==4 → No
例4: N=5, M=5, F=[1,3,4,2,5]
set(F)={1,2,3,4,5} → len=5
Q1: 5==5 → Yes Q2: 5==5 → Yes
