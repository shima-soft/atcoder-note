INF = 10**30
dist = [INF] * (N + 2)
dist[0] = 0
queue = [(0, 0)]

while queue:
    current_dist, city = heapq.heappop(queue)
    if current_dist != dist[city]:
        continue

    for next_city, cost in graph[city]:
        new_dist = current_dist + cost
        if new_dist < dist[next_city]:
            dist[next_city] = new_dist
            heapq.heappush(queue, (new_dist, next_city))
